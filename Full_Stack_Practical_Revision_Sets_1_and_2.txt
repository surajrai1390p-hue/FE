Q2 (c). TRANSFER FUNDS WITH A SEQUELIZE TRANSACTION Marks: 7 Database:
bank_db Install: npm install sequelize mysql2

CREATE DATABASE: CREATE DATABASE bank_db;

File: transfer.js Replace “your_password” with your MySQL password.

CODE: const { Sequelize, DataTypes } = require(“sequelize”);

const sequelize = new Sequelize( “bank_db”, “root”, “your_password”, {
host: “localhost”, dialect: “mysql” } );

const Account = sequelize.define(“Account”, { holderName:
DataTypes.STRING, balance: { type: DataTypes.DECIMAL(12, 2), allowNull:
false, defaultValue: 0 } });

const TransferLog = sequelize.define(“TransferLog”, { fromAccountId:
DataTypes.INTEGER, toAccountId: DataTypes.INTEGER, amount:
DataTypes.DECIMAL(12, 2) });

async function transfer(fromId, toId, amount) { const t = await
sequelize.transaction();

    try {
        if (!Number.isFinite(amount) || amount <= 0) {
            throw new Error("Amount must be greater than zero");
        }

        const sender = await Account.findByPk(fromId, {
            transaction: t,
            lock: t.LOCK.UPDATE
        });

        const receiver = await Account.findByPk(toId, {
            transaction: t,
            lock: t.LOCK.UPDATE
        });

        if (!sender || !receiver) {
            throw new Error("Account not found");
        }

        if (Number(sender.balance) < amount) {
            throw new Error("Insufficient balance");
        }

        await sender.decrement("balance", {
            by: amount,
            transaction: t
        });

        await receiver.increment("balance", {
            by: amount,
            transaction: t
        });

        await TransferLog.create({
            fromAccountId: fromId,
            toAccountId: toId,
            amount
        }, { transaction: t });

        await t.commit();
        console.log("Transfer successful");
    } catch (error) {
        await t.rollback();
        console.log("Transfer failed:", error.message);
    }

}

async function main() { try { await sequelize.authenticate(); await
sequelize.sync();

        // Accounts must already exist. Example: account 1 sends 500 to account 2.
        await transfer(1, 2, 500);
    } catch (error) {
        console.log("Database error:", error.message);
    } finally {
        await sequelize.close();
    }

}

main();

IMPORTANT SETUP: - Create two Account records before calling
transfer(). - Example account 1 balance = 2000; account 2 balance =
1000. - After transfer(1, 2, 500), balances should be 1500 and 1500. -
Do not add sample accounts automatically every time the program runs.

EXPLANATION: - A transaction groups database operations. - Both account
records are retrieved within the transaction. - decrement() debits the
sender. - increment() credits the receiver. - TransferLog records the
transfer. - commit() saves all changes. - rollback() undoes all changes
if an operation fails. - Every transfer operation must use the same
transaction object.

const app = express();

const sequelize = new Sequelize( “bank_db”, “root”, “your_password”, {
host: “localhost”, dialect: “mysql” } );

const Account = sequelize.define(“Account”, { holderName:
DataTypes.STRING, balance: DataTypes.DECIMAL(12, 2) });

const TransferLog = sequelize.define(“TransferLog”, { fromAccountId:
DataTypes.INTEGER, toAccountId: DataTypes.INTEGER, amount:
DataTypes.DECIMAL(12, 2) });

app.get(“/api/accounts/:id/statement”, async (req, res) => { const id =
Number(req.params.id);

    if (!Number.isInteger(id) || id <= 0) {
        return res.status(400).json({ message: "Invalid account ID" });
    }

    try {
        const account = await Account.findByPk(id);

        if (!account) {
            return res.status(404).json({ message: "Account not found" });
        }

        const records = await TransferLog.findAll({
            where: {
                [Op.or]: [
                    { fromAccountId: id },
                    { toAccountId: id }
                ]
            },
            order: [["createdAt", "DESC"]]
        });

        res.json({
            accountId: id,
            holderName: account.holderName,
            statement: records
        });
    } catch (error) {
        res.status(500).json({ message: "Unable to retrieve statement" });
    }

});

await sequelize.sync();

        app.listen(3000, () => {

        });
    } catch (error) {
        console.log("Database connection failed:", error.message);
    }

}

CODE: import { useEffect, useState } from “react”;

function useWindowSize() { const [size, setSize] = useState({ width:
window.innerWidth, height: window.innerHeight });

    useEffect(() => {
        function updateSize() {
            setSize({
                width: window.innerWidth,
                height: window.innerHeight
            });
        }

        window.addEventListener("resize", updateSize);

        return () => {
            window.removeEventListener("resize", updateSize);
        };
    }, []);

    return size;

}

export default useWindowSize;
